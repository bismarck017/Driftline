    89→  t: number;
    90→  top: THREE.Color;

   104→let envKeysCompiled: EnvCompiled[] | null = null;
   105→function getEnvKeys(): EnvCompiled[] {
   106→  if (!envKeysCompiled) {
   107→    envKeysCompiled = ENV_KEYS.map((k) => ({
   108→      t: k.t,
   109→      top: new THREE.Color(k.top),
   110→      hor: new THREE.Color(k.hor),
   111→      bot: new THREE.Color(k.bot),
   112→      fog: new THREE.Color(k.fog),
   113→      sunC: new THREE.Color(k.sunC),
   114→      sunI: k.sunI,
   115→      hemiS: new THREE.Color(k.hemiS),
   116→      hemiG: new THREE.Color(k.hemiG),
   117→      hemiI: k.hemiI,
   118→      night: k.night,
   119→      exp: k.exp,
   120→      water: new THREE.Color(k.water),
   121→    }));
   122→  }
   123→  return envKeysCompiled;
   124→}

   126→/** Scratch result of sampling the 24h environment. */
   127→type EnvSample = EnvCompiled;
   128→
   129→function sampleEnv(tod: number, out: EnvSample): EnvSample {
   130→  const keys = getEnvKeys();
   131→  const t = ((tod % 24) + 24) % 24;
   132→  let i = 0;
   133→  while (i < keys.length - 2 && t > keys[i + 1].t) i++;
   134→  const a = keys[i];
   135→  const b = keys[i + 1];
   136→  let u = (t - a.t) / (b.t - a.t);
   137→  u = u * u * (3 - 2 * u); // smoothstep for buttery transitions
   138→  out.t = t;
   139→  out.top.lerpColors(a.top, b.top, u);
   140→  out.hor.lerpColors(a.hor, b.hor, u);
   141→  out.bot.lerpColors(a.bot, b.bot, u);
   142→  out.fog.lerpColors(a.fog, b.fog, u);
   143→  out.sunC.lerpColors(a.sunC, b.sunC, u);
   144→  out.hemiS.lerpColors(a.hemiS, b.hemiS, u);
   145→  out.hemiG.lerpColors(a.hemiG, b.hemiG, u);
   146→  out.water.lerpColors(a.water, b.water, u);
   147→  out.sunI = lerp(a.sunI, b.sunI, u);
   148→  out.hemiI = lerp(a.hemiI, b.hemiI, u);
   149→  out.night = lerp(a.night, b.night, u);
   150→  out.exp = lerp(a.exp, b.exp, u);
   151→  return out;
   152→}

   440→  /**
   441→   * Push the current time of day into every visual system: sky gradient, sun
   442→   * and moon discs, stars, fog, hemisphere + directional lights, water tint,
   443→   * tone-mapping exposure and the car's automatic headlights.
   444→   */
   445→  private applyEnvironment() {
   446→    const e = sampleEnv(this.todHours, _env);
   447→
   448→    // sun/moon arc: 6:00 sunrise in the east, 12:00 high noon, 18:00 sunset west
   449→    const a = ((this.todHours - 6) / 24) * Math.PI * 2;
   450→    this._sunDir.set(Math.cos(a), Math.sin(a), 0.38).normalize();
   451→
   452→    (this.skyU.top.value as THREE.Color).copy(e.top);
   453→    (this.skyU.horizon.value as THREE.Color).copy(e.hor);
   454→    (this.skyU.bottom.value as THREE.Color).copy(e.bot);
   455→    (this.skyU.sunColor.value as THREE.Color).copy(e.sunC);
   456→    (this.skyU.sunDir.value as THREE.Vector3).copy(this._sunDir);

   511→        return `Key${e.key}`;

   542→  private respawn() {
   543→    const spawn = findRoadSpawnNearest(this.pos.x, this.pos.z);
   544→    this.pos.set(spawn.x, surfaceHeightAt(spawn.x, spawn.z), spawn.z);

   550→  // ── physics ────────────────────────────────────────────────────────────────

   552→  private updatePhysics(dt: number) {
   553→    const s = this.car.stats;
   554→    const k = this.keys;
   555→
   556→    // inputs
   557→    const throttle = k.has("KeyW") || k.has("ArrowUp") ? 1 : 0;
   558→    const brake = k.has("KeyS") || k.has("ArrowDown") ? 1 : 0;
   559→    const steerIn = (k.has("KeyA") || k.has("ArrowLeft") ? 1 : 0) - (k.has("KeyD") || k.has("ArrowRight") ? 1 : 0);
   560→    const handbrake = k.has("Space");
   561→
   562→    // surface conditions
   563→    const h = surfaceHeightAt(this.pos.x, this.pos.z);
   564→    const onRoad = isOnRoad(this.pos.x, this.pos.z);
   565→    const inWater = h < 0.35; // shallow water / river fords
   566→
   567→    let surfaceFactor = onRoad ? 1 : s.offroad;
   568→    if (inWater) surfaceFactor *= 0.25;
   569→    // heavy viscous drag in water — rivers can be forded slowly, lakes will bog you down
   570→    if (inWater) this.speed -= this.speed * 2.4 * dt;
   571→
   572→    // ── gearbox: auto shift up through the speed ranges, back down on lift ──
   573→    const roadTop = s.topSpeed;
   574→    const hi = this.gearTop[this.gear - 1];
   575→    const lo = this.gear > 1 ? this.gearTop[this.gear - 2] * 0.92 : 0;
   576→    if (this.speed > hi && this.gear < this.gearTop.length) this.gear++;
   577→    else if (this.speed < lo && this.gear > 1) this.gear--;
   578→
   579→    // revs within the current gear, with a little launch flare and a
   580→    // scaled "reverse gear" band when backing up
   581→    let revNorm = clamp((this.speed - lo) / Math.max(hi - lo, 1), 0, 1);
   582→    if (throttle && this.speed < 1.5) revNorm = Math.max(revNorm, 0.5);
   583→    if (this.speed < -0.3) revNorm = clamp(Math.abs(this.speed) / (roadTop * 0.25), 0, 1) * 0.85;
   584→    this.revs = lerp(this.revs, revNorm, 1 - Math.exp(-8 * dt));
   585→
   586→    // longitudinal
   587→    const topSpeed = roadTop * (onRoad ? 1 : lerp(s.offroad, 1, 0.25));
   588→    if (throttle) {
   589→      const powerCurve = 1 - Math.max(this.speed, 0) / (topSpeed * 1.35);
   590→      const torque = lerp(0.62, 1, revNorm); // short gears pull harder
   591→      this.speed += s.accel * Math.max(powerCurve, 0) * torque * surfaceFactor * dt;
   592→    }
   593→    if (brake && this.speed > 0.5) {
   594→      // heavy braking — the fronts can lock and the car skids. At speed the
   595→      // pedal overpowers the tires almost immediately (real stomp = lock);
   596→      // parking-speed stops stay controlled because lock builds slowly there.
   597→      const lockGain = (this.speed > 9 ? 3.4 : 0.9) * (onRoad ? 1 : 1.5);
   598→      this.lockProg = clamp(this.lockProg + lockGain * dt, 0, 1);
   599→      this.speed -= (this.lockProg >= 1 ? 13.5 : 26) * dt;
   600→    } else if (brake) {
   601→      this.speed -= s.accel * 0.45 * surfaceFactor * dt; // reverse
   602→    } else {
   603→      this.lockProg = Math.max(0, this.lockProg - 3 * dt);
   604→    }
   605→    this.locked = this.lockProg >= 1 && this.speed > 4;
   606→    if (handbrake) this.speed -= Math.sign(this.speed) * 14 * dt;
   607→
   608→    // drag & rolling resistance
   609→    const drag = onRoad ? 0.06 : 0.16;
   610→    this.speed -= this.speed * drag * dt;
   611→    this.speed -= Math.sign(this.speed) * (onRoad ? 0.8 : 2.2) * dt;
   612→    if (Math.abs(this.speed) < 0.02 && !throttle && !brake) this.speed = 0;
   613→    this.speed = clamp(this.speed, -topSpeed * 0.25, topSpeed);
   614→
   615→    // slope gravity: steeper climbs bleed speed, descents add it
   616→    const dirX = Math.sin(this.heading);
   617→    const dirZ = Math.cos(this.heading);
   618→    const ahead = surfaceHeightAt(this.pos.x + dirX * 2, this.pos.z + dirZ * 2);
   619→    const behind = surfaceHeightAt(this.pos.x - dirX * 2, this.pos.z - dirZ * 2);
   620→    const slope = (ahead - behind) / 4;
   621→    this.speed -= slope * 16 * dt;
   622→    // steep faces act like walls: extra braking beyond ~50% grade so the car
   623→    // can't claw its way up cliffs (this is what prevented sinking/porpoising)
   624→    if (slope > 0.5 && this.speed > 0) this.speed -= (slope - 0.5) * 120 * dt;
   625→    if (slope < -0.5 && this.speed < 0) this.speed -= (slope + 0.5) * 120 * dt;
   626→
   627→    // steering — the yaw the tires can actually deliver is capped by grip,
   628→    // so inertia limits how fast the car changes direction at speed
   629→    const speedRatio = clamp(Math.abs(this.speed) / 10, 0, 1);
   630→    const steerRate = 1.9 * s.grip * (handbrake ? 1.7 : 1);
   631→    const desired = steerIn * steerRate * speedRatio * (this.speed < 0 ? -1 : 1);
   632→    const latMax = 12.5 * s.grip * (this.locked ? 0.34 : 1) * (handbrake ? 1.55 : 1);
   633→    const yawMax = latMax / Math.max(Math.abs(this.speed), 2.5);
   634→    this.heading += clamp(desired, -yawMax, yawMax) * dt;
   635→    // locked wheels keep their momentum — the nose keeps drifting with committed
   636→    // steering input, so you counter-steer through the slide
   637→    if (this.locked) this.heading += this.steerAngle * 0.4 * clamp(this.speed / 30, 0, 1) * dt;
   638→    this.steerAngle = lerp(this.steerAngle, steerIn * 0.42, 1 - Math.exp(-10 * dt));
   639→
   640→    // torque steer: a hard launch tugs the nose to one side until speed builds
   641→    if (throttle && this.speed > -0.1) {
   642→      const decay = 1 - clamp(this.speed / 9, 0, 1);
   643→      this.heading -= (0.05 + s.accel * 0.004) * decay * decay * dt;
   644→    }
   645→
   646→    // integrate position
   647→    this.pos.x += dirX * this.speed * dt;
   648→    this.pos.z += dirZ * this.speed * dt;
   649→    this.odometer += Math.abs(this.speed) * dt;
   650→    this.throttleIn = !!throttle;
   651→
   652→    // fell into deep water? respawn (river beds carve to ~-1.9, so anything
   653→    // below -1.35 is undriveable — never leave the car floating there)
   654→    const newH = surfaceHeightAt(this.pos.x, this.pos.z);
   655→    if (newH < -1.35) {
   656→      this.respawn();
   657→      return;
   658→    }
   659→
   660→    // bogged down in shallow water (heavy drag, no traction)? after a few
   661→    // seconds of barely moving, rescue the player back onto the road
   662→    if (inWater && Math.abs(this.speed) < 0.6) {
   663→      this.stuckTimer += dt;
   664→      if (this.stuckTimer > 3) {
   665→        this.stuckTimer = 0;
   666→        this.respawn();
   667→        return;
   668→      }
   669→    } else {
   670→      this.stuckTimer = 0;
   671→    }
   672→
   673→    // ground follow with soft suspension — anti-sink:
   674→    //  · body height = average of the four wheel contact points (physical)
   675→    //  · if the terrain pokes up under the chassis centre (crest), lift to clear
   676→    //  · rise quickly (no lagging into uphills), settle down gently (airtime)
   677→    const hFL = surfaceHeightAt(this.pos.x - dirZ * 1.1 + dirX * 1.4, this.pos.z + dirX * 1.1 + dirZ * 1.4);
   678→    const hFR = surfaceHeightAt(this.pos.x + dirZ * 1.1 + dirX * 1.4, this.pos.z - dirX * 1.1 + dirZ * 1.4);
   679→    const hRL = surfaceHeightAt(this.pos.x - dirZ * 1.1 - dirX * 1.4, this.pos.z + dirX * 1.1 - dirZ * 1.4);
   680→    const hRR = surfaceHeightAt(this.pos.x + dirZ * 1.1 - dirX * 1.4, this.pos.z - dirX * 1.1 - dirZ * 1.4);
   681→    const hC = surfaceHeightAt(this.pos.x, this.pos.z);
   682→    const avg = (hFL + hFR + hRL + hRR) / 4;
   683→    const groundH = Math.max(avg, hC - 0.15);
   684→    const targetY = Math.max(groundH, -0.15);
   685→    const rate = targetY >= this.smoothY ? 30 : 13;
   686→    this.smoothY = lerp(this.smoothY, targetY, 1 - Math.exp(-rate * dt));
   687→    // hard floor: the rendered surface IS the physics surface (surfaceHeightAt
   688→    // interpolates the exact mesh triangles), so with a tight tolerance the
   689→    // wheels can visibly touch the ground everywhere — no sink, no hover
   690→    this.pos.y = Math.max(this.smoothY, Math.max(hFL, hFR, hRL, hRR, hC) - 0.1);
   691→
   692→    // body attitude from terrain (reuse the wheel samples — cheap and smooth)
   693→    const forwardPitch = Math.atan2((hFL + hFR) / 2 - (hRL + hRR) / 2, 2.8);
   694→    const sideRoll = Math.atan2((hFL + hRL) / 2 - (hFR + hRR) / 2, 2.2);
   695→    this.pitch = lerp(this.pitch, forwardPitch, 1 - Math.exp(-8 * dt));
   696→    this.roll = lerp(this.roll, sideRoll + this.steerAngle * clamp(this.speed / 40, 0, 1) * 0.18, 1 - Math.exp(-8 * dt));
   697→
   698→    // apply to car mesh
   699→    this.built.group.position.copy(this.pos);
   700→    this.built.group.rotation.set(0, 0, 0);
   701→    this.built.group.rotateY(this.heading);
   702→    this.built.group.rotateX(-this.pitch); // nose up on climb
   703→    this.built.group.rotateZ(-this.roll);
   704→
   705→    // wheels: steer + spin
   706→    this.wheelSpin += (this.speed / this.car.wheelR) * dt;
   707→    this.built.steerWheels.forEach((w) => {
   708→      w.rotation.y = this.steerAngle;
   709→    });
   710→    this.built.wheels.forEach((w) => {
   711→      w.rotation.x = -this.wheelSpin;
   712→    });
   713→  }

   717→  private updateCamera(dt: number) {
   718→    const dirX = Math.sin(this.heading);
   719→    const dirZ = Math.cos(this.heading);
   720→    const sp = Math.abs(this.speed);
   721→    const far = this.cameraMode === 1;
   722→    const dist = (far ? 16 : 8.2) + sp * (far ? 0.16 : 0.09);
   723→    const height = far ? 9.5 : 3.1 + sp * 0.012;
   724→
   725→    // reversing? the camera swings around to the front and looks down the
   726→    // car's path of travel, so you can see where you're backing into
   727→    this.revBlend = lerp(this.revBlend, this.speed < -0.6 ? 1 : 0, 1 - Math.exp(-3.2 * dt));
   728→    const side = dist * (2 * this.revBlend - 1); // -dist behind .. +dist ahead
   729→    const camX = this.pos.x + dirX * side;
   730→    const camZ = this.pos.z + dirZ * side;
   731→    const groundClear = surfaceHeightAt(camX, camZ) + 1.6;
   732→    const camY = Math.max(this.pos.y + height, groundClear);
   733→
   734→    _camPos.set(camX, camY, camZ);
   735→    this.camera.position.lerp(_camPos, 1 - Math.exp(-(far ? 3 : 6) * dt));
   736→
   737→    const lookSide = 7 * (1 - 2 * this.revBlend); // ahead .. behind
   738→    _look.set(this.pos.x + dirX * lookSide, this.pos.y + 1.6, this.pos.z + dirZ * lookSide);
   739→    this.camera.lookAt(_look);
   740→
   741→    const fov = 56 + clamp(sp / 60, 0, 1) * 10;
   742→    if (Math.abs(this.camera.fov - fov) > 0.05) {
   743→      this.camera.fov = lerp(this.camera.fov, fov, 1 - Math.exp(-4 * dt));
   744→      this.camera.updateProjectionMatrix();
   745→    }
   746→
   747→    // keep the sky dome centred on the camera
   748→    this.sky.position.copy(this.camera.position);
   749→  }

   753→  private loop = () => {
   754→    if (this.disposed) return;
   755→    this.raf = requestAnimationFrame(this.loop);
   756→    const dt = Math.min(this.clock.getDelta(), 0.05);
   757→    if (this.paused) return;
   758→
   759→    this.waterTime.value += dt;
   760→    if (this.cycleEnabled) {
   761→      this.todHours = (this.todHours + (24 / this.dayLength) * dt) % 24;
   762→    }
   763→    this.applyEnvironment();
   764→    this.updatePhysics(dt);
   765→    this.tiles.update(this.pos.x, this.pos.z);
   766→
   767→    // shadow frustum follows the car (snapped to reduce shimmer); the light
   768→    // itself comes from the sun by day and the moon by night
   769→    const sx = Math.round(this.pos.x / 2) * 2;
   770→    const sz = Math.round(this.pos.z / 2) * 2;
   771→    _lightDir.copy(this._sunDir);
   772→    if (_lightDir.y <= 0) _lightDir.multiplyScalar(-1); // moonlight
   773→    _lightDir.y = Math.max(Math.abs(_lightDir.y), 0.3); // keep shadows well-conditioned
   774→    _lightDir.normalize();
   775→    this.sun.position.set(sx + _lightDir.x * 150, _lightDir.y * 150, sz + _lightDir.z * 150);
   776→    this.sun.target.position.set(sx, 0, sz);
   777→
   778→    this.updateCamera(dt);
   779→    this.renderer.render(this.scene, this.camera);
   780→
   781→    // HUD ~16x per second (keeps the rev bar lively)
   782→    this.hudTimer += dt;
   783→    if (this.hudTimer > 0.06 && this.onHud) {
   784→      this.hudTimer = 0;
   785→      const h = surfaceHeightAt(this.pos.x, this.pos.z);
   786→      const tt = h < 0.35 && !isOnRoad(this.pos.x, this.pos.z) ? "ocean" : terrainTypeAt(this.pos.x, this.pos.z, Math.max(h, 0));
   787→      if (typeof window !== "undefined" && (window as unknown as Record<string, unknown>).__dlDebug) {
   788→        console.log(
   789→          `[dl] x=${this.pos.x.toFixed(1)} z=${this.pos.z.toFixed(1)} carY=${this.pos.y.toFixed(2)} terrain=${h.toFixed(2)} agl=${(this.pos.y - h).toFixed(2)}`
   790→        );
   791→      }
   792→      const hh = Math.floor(this.todHours);
   793→      const mm = Math.floor((this.todHours - hh) * 60);
   794→      const clock = `${String(hh).padStart(2, "0")}:${String(mm).padStart(2, "0")}`;
   795→      this.onHud({
   796→        kmh: Math.round(Math.abs(this.speed) * 3.6),
   797→        biome: BIOME_LABEL[tt] ?? tt,
   798→        x: this.pos.x,
   799→        z: this.pos.z,
   800→        tilesLoaded: this.tiles.loadedCount,
   801→        heading: this.heading,
   802→        onRoad: isOnRoad(this.pos.x, this.pos.z),
   803→        clock,
   804→        tod: this.todHours,
   805→        gear: this.speed < -0.3 ? -1 : Math.abs(this.speed) < 0.3 && !this.throttleIn ? 0 : this.gear,
   806→        revs: this.revs,
   807→        distance: this.odometer,
   808→        locked: this.locked,
   809→      });
   810→    }
   811→  };

   819→/** Spawn on the road lattice closest to a given position (for the R-key reset). */
   820→function findRoadSpawnNearest(px: number, pz: number): { x: number; z: number; heading: number } {
   821→  const fallback = findRoadSpawn();
   822→  let best = fallback;
   823→  let bestD = Infinity;
   824→  // coarse scan around the player, then refine
   825→  for (let z = pz - 480; z <= pz + 480; z += 48) {
   826→    let bx = px;
   827→    let bd = Infinity;
   828→    for (let dx = -300; dx <= 300; dx += 6) {
   829→      const d = roadDistance(px + dx, z);
   830→      if (d < bd) {
   831→        bd = d;
   832→        bx = px + dx;
   833→      }
   834→    }
   835→    if (bd > 6 || surfaceHeightAt(bx, z) < 1.2) continue;
   836→    const x2 = bestNextX(bx, z + 1);
   837→    const cand = { x: bx, z, heading: Math.atan2(x2 - bx, 1) };
   838→    const d = (bx - px) ** 2 + (z - pz) ** 2;
   839→    if (d < bestD) {
   840→      bestD = d;
   841→      best = cand;
   842→    }
   843→  }
   844→  return bestD < Infinity ? best : fallback;
   845→}
   846→
   847→function bestNextX(bx: number, z: number): number {
   848→  let best = bx;
   849→  let bd = Infinity;
   850→  for (let dx = -4; dx <= 4; dx += 2) {
   851→    const d = roadDistance(bx + dx, z);
   852→    if (d < bd) {
   853→      bd = d;
   854→      best = bx + dx;
   855→    }
   856→  }
   857→  return best;
   858→}
   859→
